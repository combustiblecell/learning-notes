沿用原书记号。后文把 $\mathbf{x}$ 写成 $\bm{x}$，含义不变。

\paragraph{数字}
\begin{align*}
x&\in\mathbb{R}
&&\text{标量},\\
\bm{x}&\in\mathbb{R}^{n}
&&\text{向量},\\
\bm{X}&\in\mathbb{R}^{a\times b}
&&\text{矩阵},\\
\mathsf{X}&&&\text{高阶张量},\\
\bm{I}&&&\text{单位矩阵},\\
x_i=[\bm{x}]_i&&&\text{向量第 $i$ 个元素},\\
x_{ij}=[\bm{X}]_{ij}&&&\text{矩阵第 $i$ 行第 $j$ 列}.
\end{align*}
